// DDS信号发生器核心模块
module DDS_Core(
    input           clk,         // 系统时钟 100MHz
    input           rst_n,       // 复位信号（低电平有效）
    input [31:0]    fcw,         // 频率控制字
    input [1:0]     wave_sel,    // 波形选择
    output reg [7:0] dds_out     // 8位输出
);

reg [31:0] phase_acc;      // 32位相位累加器
reg [7:0]  sin_lut [0:255]; // 正弦波查找表

// 1. 相位累加器
always @(posedge clk or negedge rst_n) begin
    if(!rst_n)
        phase_acc <= 32'd0;
    else
        phase_acc <= phase_acc + fcw;
end

// 2. 正弦波查找表
wire [7:0] sin_lut_out = sin_lut[phase_acc[31:24]];

// 3. 方波生成
wire [7:0] square_wave = (phase_acc[31:24] > 8'd127) ? 8'd255 : 8'd0;

// 4. 三角波生成
wire [7:0] triangle_wave = (phase_acc[31:24] <= 8'd127) ?
    (phase_acc[31:24] << 1) :
    (8'd255 - ((phase_acc[31:24] - 8'd128) << 1));

// 5. 波形选择
always @(posedge clk) begin
    case(wave_sel)
        2'b00: dds_out <= sin_lut_out;
        2'b01: dds_out <= square_wave;
        2'b10: dds_out <= triangle_wave;
    endcase
end

endmodule
